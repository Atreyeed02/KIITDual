UDP CLIENT-
#include<stdio.h>
#include<unistd.h>
#include<netinet/in.h>
#include<arpa/inet.h>
#include<sys/socket.h>

typedef struct student{
    int roll;
    char name[25];
    float cgpa;
} student;

int main()
{
    int sockfd, sendt, recvf;
    struct sockaddr_in server;

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    if(sockfd < 0)
    {
        printf("Socket Creation Failed");
        return 0;
    }

    server.sin_family = AF_INET;
    server.sin_port = htons(8080);
    server.sin_addr.s_addr = inet_addr("127.0.0.1");

    student s1;

    printf("Enter student details - roll, name, cgpa: ");

    scanf("%d", &s1.roll);
    scanf("%24s", s1.name);
    scanf("%f", &s1.cgpa);

    sendt = sendto(sockfd, &s1, sizeof(s1), 0,
                   (struct sockaddr*)&server, sizeof(server));

    if(sendt < 0)
    {
        printf("Send failed");
    }
    else
    {
        printf("Student details sent");
    }

    close(sockfd);
    return 0;
}


UDP SERVER-

#include<stdio.h>
#include<sys/socket.h>
#include<arpa/inet.h>
#include<netinet/in.h>
#include<unistd.h>

typedef struct Student{
    int roll;
    char name[25];
    float cgpa;
} student;

int main()
{
    int sockfd, sendt, recvf;
    struct sockaddr_in server, cli;

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    if(sockfd < 0)
    {
        printf("Socket Creation Failed");
        return 0;
    }

    server.sin_family = AF_INET;
    server.sin_port = htons(8080);
    server.sin_addr.s_addr = inet_addr("127.0.0.1");

    int bd = bind(sockfd, (struct sockaddr*)&server,
                 sizeof(server));

    if(bd < 0)
    {
        printf("Bind failed");
        return 0;
    }

    student s1;

    int s = sizeof(cli);

    recvf = recvfrom(sockfd, &s1, sizeof(s1), 0,
                    (struct sockaddr*)&cli, &s);

    if(recvf < 0)
    {
        printf("Receive failed");
    }
    else
    {
        printf("\nName of student: %s", s1.name);
        printf("\nRoll of student: %d", s1.roll);
        printf("\nCGPA of student: %.2f", s1.cgpa);
    }

    close(sockfd);
    return 0;
}
